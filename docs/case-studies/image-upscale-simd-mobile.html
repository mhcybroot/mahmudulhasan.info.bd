Contrast-Adaptive Kernel

src/kernel/simd_cas.rs #[target_feature(enable = "avx2")] 
// SIMD 8-lane contrast-adaptive sharpening vector
#[inline(always)]
pub unsafe fn apply_cas_kernel(
    src: &[f32], 
    dst: &mut [f32], 
    stride: usize,
    sharpness: f32
) {
    // Broadcast CAS sharpness coefficient across 256 bits
    let w_coeff = _mm256_set1_ps(-0.125 * sharpness);
    let one_vec = _mm256_set1_ps(1.0);

    // Unrolled 8-pixel SIMD stride loop
    for x in (0..stride).step_by(8) {
        let p_center = _mm256_loadu_ps(src.as_ptr().add(x));
        let p_top    = _mm256_loadu_ps(src.as_ptr().add(x - stride));
        let p_bottom = _mm256_loadu_ps(src.as_ptr().add(x + stride));
        let p_left   = _mm256_loadu_ps(src.as_ptr().add(x - 1));
        let p_right  = _mm256_loadu_ps(src.as_ptr().add(x + 1));

        // Local contrast min/max boundary detection
        let min_edge = _mm256_min_ps(p_center, 
            _mm256_min_ps(p_top, _mm256_min_ps(p_bottom, 
            _mm256_min_ps(p_left, p_right))));
        let max_edge = _mm256_max_ps(p_center, 
            _mm256_max_ps(p_top, _mm256_max_ps(p_bottom, 
            _mm256_max_ps(p_left, p_right))));

        // Fused multiply-accumulate to eliminate halo overshoot
        let neighbors = _mm256_add_ps(_mm256_add_ps(p_top, p_bottom), 
                                      _mm256_add_ps(p_left, p_right));
        let filtered  = _mm256_fmadd_ps(neighbors, w_coeff, p_center);
        let clamped   = _mm256_max_ps(min_edge, _mm256_min_ps(max_edge, filtered));

        _mm256_storeu_ps(dst.as_mut_ptr().add(x), clamped);
    }
}
